cmake_minimum_required(VERSION 3.25)
project(PickpocketingLoot VERSION 1.1.5 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ENABLE_SKYRIM_SE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(COMMONLIB_ENABLE_IPO OFF CACHE BOOL "" FORCE)
set(SKSE_SUPPORT_PATCH_SAFETY OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/CommonLibSSE-NG)
include(vendor/CommonLibSSE-NG/cmake/CommonLibSSE.cmake)
add_commonlibsse_plugin(PickpocketingLoot NAME "Dynamic Pickpocket Loot" AUTHOR "Maxmc" VERSION 1.1.5 SOURCES src/Plugin.cpp)
target_include_directories(PickpocketingLoot PRIVATE src)
target_precompile_headers(PickpocketingLoot PRIVATE src/PCH.h)
target_compile_definitions(PickpocketingLoot PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(PickpocketingLoot PRIVATE /W4 /permissive-)
set_target_properties(PickpocketingLoot PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/package/SKSE/Plugins")
enable_testing()
add_executable(CoreTests tests/CoreTests.cpp)
target_include_directories(CoreTests PRIVATE src)
add_test(NAME CoreBehavior COMMAND CoreTests "${CMAKE_CURRENT_SOURCE_DIR}/config/DynamicPickpocketLoot.ini")

# Mirror the installed layout for local build inspection.
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/package/SKSE/Plugins")
configure_file(config/DynamicPickpocketLoot.ini "${CMAKE_BINARY_DIR}/package/SKSE/Plugins/DynamicPickpocketLoot.ini" COPYONLY)
